cmake_minimum_required(VERSION 3.24)
project(wpe64 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_library(wpe64-sqlite STATIC third_party/sqlite/sqlite3.c)
target_compile_definitions(wpe64-sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
add_library(wpe64-common STATIC src/common/ipc_codec.cpp src/common/ipc_frame.cpp src/common/ipc_pipe.cpp src/common/packet_frame.cpp src/common/packet_ring.cpp)
target_include_directories(wpe64-common PUBLIC src)
target_link_libraries(wpe64-common PUBLIC advapi32)
if(MSVC)
  add_compile_options(/W4 /WX /permissive- /utf-8)
  target_compile_options(wpe64-common PRIVATE /W4 /WX /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  target_compile_options(wpe64-common PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(wpe64-bridge STATIC src/shell/web_bridge.cpp)
target_include_directories(wpe64-bridge PUBLIC src)
target_include_directories(wpe64-bridge SYSTEM PUBLIC third_party)
add_library(wpe64-data STATIC src/shell/database.cpp src/shell/data_service.cpp src/shell/data_edit.cpp src/shell/data_config_lists.cpp src/shell/editor_xml.cpp src/shell/data_worker.cpp src/shell/xml_document.cpp src/shell/xml_crypto.cpp src/shell/config_xml.cpp src/shell/data_files.cpp)
target_compile_definitions(wpe64-data PRIVATE _WIN32_WINNT=0x0600)
target_link_libraries(wpe64-data PUBLIC wpe64-bridge wpe64-sqlite ole32 shlwapi xmllite advapi32 bcrypt iphlpapi ws2_32)
add_library(wpe64-clipboard STATIC src/shell/clipboard.cpp src/shell/clipboard_worker.cpp)
target_include_directories(wpe64-clipboard PUBLIC src)
target_link_libraries(wpe64-clipboard PUBLIC user32)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_executable(wpe64-app WIN32 src/shell/main.cpp src/shell/app.manifest)
  target_link_libraries(wpe64-app PRIVATE wpe64-data wpe64-clipboard
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/x64/WebView2LoaderStatic.lib"
    ole32 oleaut32 user32 shell32 shlwapi advapi32 dwmapi version)
  target_include_directories(wpe64-app SYSTEM PRIVATE third_party/webview2/include)
  target_compile_definitions(wpe64-app PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
endif()
if(BUILD_TESTING)
  add_executable(wpe64-codec-test tests/codec_test.cpp)
  target_link_libraries(wpe64-codec-test PRIVATE wpe64-common)
  add_test(NAME codec-smoke COMMAND wpe64-codec-test)
  add_executable(wpe64-parity-test tests/protocol_parity_test.cpp)
  target_link_libraries(wpe64-parity-test PRIVATE wpe64-common)
  add_test(NAME protocol-edge COMMAND wpe64-parity-test)
  add_executable(wpe64-ring-test tests/packet_ring_test.cpp)
  target_link_libraries(wpe64-ring-test PRIVATE wpe64-common)
  add_test(NAME packet-ring COMMAND wpe64-ring-test)
  add_executable(wpe64-pipe-test tests/ipc_pipe_test.cpp)
  target_link_libraries(wpe64-pipe-test PRIVATE wpe64-common)
  add_test(NAME named-pipe-transport COMMAND wpe64-pipe-test)
  add_executable(wpe64-bridge-test tests/web_bridge_test.cpp)
  target_link_libraries(wpe64-bridge-test PRIVATE wpe64-bridge)
  add_test(NAME web-bridge COMMAND wpe64-bridge-test)
  add_executable(wpe64-data-test tests/data_service_test.cpp)
  target_link_libraries(wpe64-data-test PRIVATE wpe64-data)
  add_test(NAME native-data COMMAND wpe64-data-test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/filter-edit.json")
  add_executable(wpe64-data-worker-test tests/data_worker_test.cpp)
  target_link_libraries(wpe64-data-worker-test PRIVATE wpe64-data)
  add_test(NAME data-worker COMMAND wpe64-data-worker-test)
  add_executable(wpe64-editor-test tests/editor_test.cpp)
  target_link_libraries(wpe64-editor-test PRIVATE wpe64-data)
  add_test(NAME editor-data COMMAND wpe64-editor-test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")

  target_link_libraries(wpe64-config-file-test PRIVATE wpe64-data)
  add_test(NAME configuration-files COMMAND wpe64-config-file-test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_executable(wpe64-clipboard-test tests/clipboard_test.cpp)
  target_link_libraries(wpe64-clipboard-test PRIVATE wpe64-clipboard)
  add_test(NAME native-clipboard COMMAND wpe64-clipboard-test)
