cmake_minimum_required(VERSION 3.24)
project(wpe64 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_library(wpe64-sqlite STATIC third_party/sqlite/sqlite3.c)
target_compile_definitions(wpe64-sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
add_library(wpe64-common STATIC src/common/ipc_codec.cpp src/common/ipc_frame.cpp src/common/ipc_pipe.cpp src/common/ipc_session.cpp src/common/packet_frame.cpp src/common/packet_ring.cpp)
target_include_directories(wpe64-common PUBLIC src)
target_link_libraries(wpe64-common PUBLIC advapi32)
if(MSVC)
  add_compile_options(/W4 /WX /permissive- /utf-8)
  target_compile_options(wpe64-common PRIVATE /W4 /WX /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  target_compile_options(wpe64-common PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set(WPE64_MINHOOK_SOURCES
  third_party/minhook/src/buffer.c
  third_party/minhook/src/hook.c
  third_party/minhook/src/trampoline.c)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  list(APPEND WPE64_MINHOOK_SOURCES third_party/minhook/src/hde/hde64.c)
else()
  list(APPEND WPE64_MINHOOK_SOURCES third_party/minhook/src/hde/hde32.c)
endif()
add_library(wpe64-minhook STATIC ${WPE64_MINHOOK_SOURCES})
target_include_directories(wpe64-minhook PUBLIC third_party/minhook/include)
target_include_directories(wpe64-minhook PRIVATE third_party/minhook/src third_party/minhook/src/hde)
if(MSVC)
  target_compile_options(wpe64-minhook PRIVATE /W0)
endif()
add_library(wpe64-hook-runtime STATIC src/target/hook_manager.cpp src/target/filter_engine.cpp src/target/winsock_hook.cpp)
target_include_directories(wpe64-hook-runtime PUBLIC src)
target_link_libraries(wpe64-hook-runtime PUBLIC wpe64-minhook wpe64-common ws2_32)
add_library(wpe64-target-core STATIC src/target/headless_core.cpp)
target_include_directories(wpe64-target-core PUBLIC src)
target_link_libraries(wpe64-target-core PUBLIC wpe64-common)
add_library(wpe64-hook SHARED src/target/dllmain.cpp src/target/injected_runtime.cpp)
set_target_properties(wpe64-hook PROPERTIES PREFIX "" OUTPUT_NAME "wpe64-hook")
target_include_directories(wpe64-hook PUBLIC src)
target_link_libraries(wpe64-hook PRIVATE wpe64-target-core wpe64-hook-runtime)
add_library(wpe64-bridge STATIC src/shell/web_bridge.cpp)
target_include_directories(wpe64-bridge PUBLIC src)
target_include_directories(wpe64-bridge SYSTEM PUBLIC third_party)
add_library(wpe64-data STATIC src/shell/database.cpp src/shell/data_service.cpp src/shell/data_edit.cpp src/shell/data_config_lists.cpp src/shell/editor_xml.cpp src/shell/data_worker.cpp src/shell/xml_document.cpp src/shell/xml_crypto.cpp src/shell/config_xml.cpp src/shell/data_files.cpp)
target_compile_definitions(wpe64-data PRIVATE _WIN32_WINNT=0x0600)
target_link_libraries(wpe64-data PUBLIC wpe64-bridge wpe64-common wpe64-sqlite ole32 shlwapi xmllite advapi32 bcrypt iphlpapi ws2_32)
add_library(wpe64-clipboard STATIC src/shell/clipboard.cpp src/shell/clipboard_worker.cpp)
target_include_directories(wpe64-clipboard PUBLIC src)
target_link_libraries(wpe64-clipboard PUBLIC user32)
add_library(wpe64-injector STATIC src/shell/process_injector.cpp)
target_include_directories(wpe64-injector PUBLIC src)
target_link_libraries(wpe64-injector PUBLIC wpe64-common)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_executable(wpe64-app WIN32 src/shell/main.cpp src/shell/app.manifest)
  target_link_libraries(wpe64-app PRIVATE wpe64-data wpe64-clipboard
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/x64/WebView2LoaderStatic.lib"
    ole32 oleaut32 user32 shell32 shlwapi advapi32 dwmapi version)
  target_include_directories(wpe64-app SYSTEM PRIVATE third_party/webview2/include)
  target_compile_definitions(wpe64-app PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
endif()
