cmake_minimum_required(VERSION 3.24)
project(wpe64 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_library(wpe64-sqlite STATIC third_party/sqlite/sqlite3.c)
target_compile_definitions(wpe64-sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
add_library(wpe64-common STATIC src/common/ipc_codec.cpp src/common/ipc_frame.cpp src/common/ipc_pipe.cpp src/common/ipc_session.cpp src/common/packet_frame.cpp src/common/packet_ring.cpp)
target_include_directories(wpe64-common PUBLIC src)
target_link_libraries(wpe64-common PUBLIC advapi32)
if(MSVC)
  add_compile_options(/W4 /WX /permissive- /utf-8)
  target_compile_options(wpe64-common PRIVATE /W4 /WX /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  target_compile_options(wpe64-common PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(MSVC)
  target_compile_options(wpe64-minhook PRIVATE /W0)
endif()

target_include_directories(wpe64-hook PUBLIC src)

if(WIN32)
  # This target is deliberately excluded from the default build.  It must be
  # generated with a Win32 generator and shipped beside the x64 shell so the
  # shell can inject into WOW64 targets without crossing architectures.
  add_executable(wpe64-x86-helper EXCLUDE_FROM_ALL src/shell/x86_helper_main.cpp)
  target_link_libraries(wpe64-x86-helper PRIVATE wpe64-injector)
  set_target_properties(wpe64-x86-helper PROPERTIES OUTPUT_NAME "wpe64-x86-helper")
endif()
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  add_executable(wpe64-bootstrap WIN32 src/shell/bootstrap_main.cpp src/shell/app.rc)
  target_link_libraries(wpe64-bootstrap PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/x64/WebView2LoaderStatic.lib"
    ole32 oleaut32 user32 shell32 shlwapi advapi32 version)
  target_include_directories(wpe64-bootstrap SYSTEM PRIVATE third_party/webview2/include)
  target_compile_definitions(wpe64-bootstrap PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)

  add_executable(wpe64-app WIN32 src/shell/main.cpp src/shell/target_link.cpp src/shell/app.manifest src/shell/app.rc)
  target_link_libraries(wpe64-app PRIVATE wpe64-data wpe64-clipboard wpe64-injector
    wpe64-proxy-runtime
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/x64/WebView2LoaderStatic.lib"
    ole32 oleaut32 user32 shell32 shlwapi advapi32 dwmapi version wininet)
  target_include_directories(wpe64-app SYSTEM PRIVATE third_party/webview2/include)
  target_compile_definitions(wpe64-app PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  add_dependencies(wpe64-app wpe64-hook)
  # Keep a directly runnable Release/Debug folder self-contained.  The
  # executable's default arguments look beside it for wwwroot and the hook;
  # copying both at build time prevents the common blank-window / missing-DLL
  # failure when someone double-clicks wpe64-app.exe.
  add_custom_command(TARGET wpe64-app POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CMAKE_CURRENT_SOURCE_DIR}/wwwroot" "$<TARGET_FILE_DIR:wpe64-app>/wwwroot"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:wpe64-hook>" "$<TARGET_FILE_DIR:wpe64-app>/wpe64-hook.dll")
endif()
